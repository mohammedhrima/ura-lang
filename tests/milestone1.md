## 010 — bool: bool parameters and return values

```ura
// expect: 3
fn is_even(n i32) bool:
    return n % 2 == 0

fn both(a bool, b bool) bool:
    return a and b

fn main() i32:
    n i32 = 0
    if is_even(4):
        n += 1
    if both(is_even(2), is_even(6)):
        n += 2
    if both(is_even(2), is_even(7)):
        n += 10
    return n
```

## 248 — calls: arguments over several lines

```ura
// expect: 6
fn add(a i32, b i32, c i32) i32:
    return a + b + c

fn main() i32:
    return add(
        1,
        2,
    3)
```

### llvm ir

```llvm
; ModuleID = 'ura'
source_filename = "ura"
target datalayout = "e-m:e-p270:32:32-p271:32:32-p272:64:64-i64:64-f80:128-n8:16:32:64-S128"
target triple = "x86_64-unknown-linux-gnu"

define i32 @add(i32 %0, i32 %1, i32 %2) {
entry:
  %a = alloca i32, align 4
  %b = alloca i32, align 4
  %c = alloca i32, align 4
  store i32 %0, i32* %a, align 4
  store i32 %1, i32* %b, align 4
  store i32 %2, i32* %c, align 4
  %"$a" = load i32, i32* %a, align 4
  %"$b" = load i32, i32* %b, align 4
  %ADD = add i32 %"$a", %"$b"
  %"$c" = load i32, i32* %c, align 4
  %ADD1 = add i32 %ADD, %"$c"
  ret i32 %ADD1
}

define i32 @main() {
entry:
  %call.add = call i32 @add(i32 1, i32 2, i32 3)
  ret i32 %call.add
}
```

## 249 — variables: a number literal takes the type of its i8 target

```ura
// expect: 1
fn main() i32:
    c i8 = 5
    if c == 5:
        return 1
    return 0
```

### llvm ir

```llvm
; ModuleID = 'ura'
source_filename = "ura"
target datalayout = "e-m:e-p270:32:32-p271:32:32-p272:64:64-i64:64-f80:128-n8:16:32:64-S128"
target triple = "x86_64-unknown-linux-gnu"

define i32 @main() {
entry:
  %c = alloca i8, align 1
  store i8 5, i8* %c, align 1
  br label %if0

if0:                                              ; preds = %entry
  %"$c" = load i8, i8* %c, align 1
  %EQ = icmp eq i8 %"$c", 5
  br i1 %EQ, label %if0.then, label %endif

if0.then:                                         ; preds = %if0
  ret i32 1

endif:                                            ; preds = %if0
  ret i32 0
}
```
